import sys, io, re
root = sys.argv[1]
files = {}
def get(rel):
    if rel not in files:
        files[rel] = io.open(root + '/' + rel, encoding='utf-8', newline='').read()
    return files[rel]
def rep(rel, old, new, count=1):
    s = get(rel)
    n = s.count(old)
    if n != count:
        sys.exit("ANCHOR FAIL %s (found %d, want %d): %r" % (rel, n, count, old[:80]))
    files[rel] = s.replace(old, new)

	var clean := raw.replace("\\n", " ").replace("\\r", " ").replace("\\t", " ").strip_edges()
	return clean.substr(0, PLAYER_NAME_MAX_LENGTH)

# Nur der selbst eingegebene Name (kann leer sein).
func get_custom_player_name(index: int) -> String:
	if index < 1 or index > MAX_PLAYERS or is_bot_slot(index):
		return ""
	return player_names[index - 1]

# Anzeigename: eingegebener Benutzername oder Ersatz wie "Spieler 2" / "Bot 3".
func get_player_name(index: int) -> String:
	var custom := get_custom_player_name(index)
	if not custom.is_empty():
		return custom
	return ("Bot %d" if is_bot_slot(index) else "Spieler %d") % index

		client_player_name.rpc_id(1, player_name)

@rpc("any_peer", "call_remote", "reliable")
func client_player_name(player_name: String) -> void:
	if not is_host:
		return
	var slot := get_slot_for_peer(multiplayer.get_remote_sender_id())
	if slot <= 0:
		return
	Global.set_player_name(slot, player_name)
