import sys, io, re
root = sys.argv[1]
files = {}
def get(rel):
    if rel not in files:
        files[rel] = io.open(root + '/' + rel, encoding='utf-8', newline='').read()
    return files[rel]
def rep(rel, old, new, count=1):
    s = get(rel); n = s.count(old)
    if n != count:
        sys.exit("ANCHOR FAIL %s (found %d, want %d): %r" % (rel, n, count, old[:80]))
    files[rel] = s.replace(old, new)

func _spawn_dust(local_pos: Vector2, direction: float) -> void:
	if not animate:
		return
	_dust.append({
		"pos": local_pos,
		"vel": Vector2(direction * 14.0, -6.0),
		"life": DUST_DURATION,
	})
	while _dust.size() > 10:
		_dust.remove_at(0)

func _update_dust(delta: float) -> void:
	# Staub bleibt in der Welt liegen, statt mit der Figur mitzuwandern.
	var world_scale := maxf(0.01, absf(get_global_transform().get_scale().x))
	for i in range(_dust.size() - 1, -1, -1):
		var puff: Dictionary = _dust[i]
		var life := float(puff["life"]) - delta
		if life <= 0.0:
			_dust.remove_at(i)
			continue
		var velocity: Vector2 = puff["vel"]
		velocity *= 0.9
		var position: Vector2 = puff["pos"]
		position += velocity * delta - _motion_velocity * delta / world_scale
		puff["life"] = life
		puff["vel"] = velocity
		puff["pos"] = position
		_dust[i] = puff
